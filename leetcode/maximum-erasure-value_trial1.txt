class Solution:
    def maximumUniqueSubarray(self, nums: List[int]) -> int:
        seen = set()
        running_sum = 0
        l = 0
        max_sum = 0
        for r in range(len(nums)):
            while nums[r] in seen:
                seen.remove(nums[l])
                running_sum -= nums[l]
                l += 1
            running_sum += nums[r]
            seen.add(nums[r])
            max_sum = max(max_sum, running_sum)
        
        return max_sum

class Solution:
    def maximumSubarraySum(self, nums: List[int], k: int) -> int:
        count = defaultdict(int)
        left = 0
        current_sum = 0
        max_sum = 0
        
        for right in range(len(nums)):
            count[nums[right]] += 1
            current_sum += nums[right]
            
            if right - left + 1 > k:
                count[nums[left]] -= 1
                current_sum -= nums[left]
                if count[nums[left]] == 0:
                    del count[nums[left]]
                left += 1

            if len(count) == k:
                max_sum = max(max_sum, current_sum)
        
        return max_sum

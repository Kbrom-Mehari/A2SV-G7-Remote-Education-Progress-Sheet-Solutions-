class Solution:
    def maxWidthOfVerticalArea(self, points: List[List[int]]) -> int:
        x_coordinates = []
        gap = 0
        for point in points:
            x_coordinates.append(point[0])
        x_coordinates.sort()
        for i in range(1, len(x_coordinates)):
            gap = max(gap, x_coordinates[i] - x_coordinates[i-1])
        return gap

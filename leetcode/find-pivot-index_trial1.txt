class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        n = len(nums)
        running_sum = 0
        prefix_sum = []
        for i in range(n):
            prefix_sum.append(running_sum + nums[i])
            running_sum += nums[i]
        for i in range(n):
            left_sum = (0 if i == 0 else prefix_sum[i-1])
            right_sum = prefix_sum[n-1] - prefix_sum[i]
            if left_sum == right_sum:
                return i
        return -1

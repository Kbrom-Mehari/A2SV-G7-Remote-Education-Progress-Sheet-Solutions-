class Solution:
    def dividePlayers(self, skill: List[int]) -> int:
        n = len(skill)
        teams = n // 2
        total = sum(skill)
        target = total // teams
        start = 0
        end = n - 1
        chemistry = 0
        if total % teams != 0:
            return -1
        skill.sort()
        while start < end:
            if skill[start] + skill[end] != target:
                return -1
            chemistry += skill[start] * skill[end]
            start += 1
            end -= 1
        return chemistry
